cmake_minimum_required(VERSION 3.16)
project(adaptive_compliance_controller)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)
find_package(force_impedance_controller REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
add_library(${PROJECT_NAME} SHARED src/adaptive_compliance_controller.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} force_impedance_controller realtime_tools pluginlib std_msgs Eigen3)
pluginlib_export_plugin_description_file(controller_interface adaptive_compliance_controller_plugin.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md NOTICE LICENSE DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/estimate_tissue.py DESTINATION lib/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_export_dependencies(force_impedance_controller realtime_tools pluginlib std_msgs Eigen3)
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME tissue_estimator_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_estimator.py)
  add_executable(qp_probe test/qp_probe.cpp)
  target_include_directories(qp_probe PRIVATE include)
  target_link_libraries(qp_probe Eigen3::Eigen)
  add_executable(adaptive_math_test test/adaptive_math_test.cpp)
  target_include_directories(adaptive_math_test PRIVATE include)
  target_link_libraries(adaptive_math_test Eigen3::Eigen)
  add_test(NAME adaptive_math_test COMMAND adaptive_math_test)
  add_executable(adaptive_lifecycle_test test/adaptive_lifecycle_test.cpp)
  target_link_libraries(adaptive_lifecycle_test ${PROJECT_NAME})
  ament_target_dependencies(adaptive_lifecycle_test force_impedance_controller realtime_tools pluginlib std_msgs)
  add_test(NAME adaptive_lifecycle_test COMMAND adaptive_lifecycle_test)
endif()
ament_package()
